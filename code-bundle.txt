"use client";

import { useActionState, useEffect, useRef, type ReactNode } from "react";
import { Loader2 } from "lucide-react";
import { Button } from "@/components/ui";
import type { ActionState } from "@/lib/types";

type Action = (prev: ActionState, formData: FormData) => Promise<ActionState>;

/** Wraps a form around a Server Action with pending, error and success feedback. */
export function AdminForm({
  action,
  children,
  submitLabel = "Save changes",
  resetOnSuccess = false,
  className,
  hideSubmit = false,
  danger = false,
  confirmMessage,
}: {
  action: Action;
  children: ReactNode;
  submitLabel?: string;
  resetOnSuccess?: boolean;
  className?: string;
  hideSubmit?: boolean;
  danger?: boolean;
  /** When set, the browser asks for confirmation before submitting (use for deletes). */
  confirmMessage?: string;
}) {
  const [state, formAction, pending] = useActionState(action, null);
  const ref = useRef<HTMLFormElement>(null);
  useEffect(() => {
    if (state?.ok && resetOnSuccess) ref.current?.reset();
  }, [state, resetOnSuccess]);

  return (
    <form
      ref={ref}
      action={formAction}
      className={className ?? "space-y-5"}
      noValidate
      onSubmit={(e) => {
        if (confirmMessage && !window.confirm(confirmMessage)) e.preventDefault();
      }}
    >
      {children}
      <div aria-live="polite">
        {state?.message && (
          <p
            role={state.ok ? "status" : "alert"}
            className={
              state.ok
                ? "rounded-lg border border-green-200 bg-green-50 px-3 py-2 text-sm font-medium text-success"
                : "rounded-lg border border-danger/40 bg-red-50 px-3 py-2 text-sm font-medium text-danger"
            }
          >
            {state.message}
          </p>
        )}
        {state && !state.ok && state.errors && Object.keys(state.errors).length > 0 && (
          <ul role="alert" className="mt-2 list-disc space-y-0.5 pl-5 text-sm text-danger">
            {Object.entries(state.errors).map(([field, msg]) => (
              <li key={field}>
                <span className="font-semibold">{field.replace(/_/g, " ")}:</span> {msg}
              </li>
            ))}
          </ul>
        )}
      </div>
      {!hideSubmit && (
        <Button type="submit" disabled={pending} variant={danger ? "danger" : "primary"}>
          {pending ? (
            <>
              <Loader2 className="h-4 w-4 animate-spin" aria-hidden="true" /> Savingâ€¦
            </>
          ) : (
            submitLabel
          )}
        </Button>
      )}
    </form>
  );
}

/** Server-error-aware field wrapper. `errors` comes from the action result via <FormErrors>. */
export function Field({
  label,
  name,
  hint,
  children,
}: {
  label: string;
  name: string;
  hint?: string;
  children: ReactNode;
}) {
  return (
    <div>
      <label htmlFor={name} className="mb-1.5 block text-sm font-semibold text-charcoal">
        {label}
      </label>
      {children}
      {hint && <p className="mt-1 text-xs text-ink-soft">{hint}</p>}
    </div>
  );
}

export function Check({
  name,
  label,
  defaultChecked,
  hint,
}: {
  name: string;
  label: string;
  defaultChecked?: boolean;
  hint?: string;
}) {
  return (
    <label className="flex items-start gap-3 text-sm">
      <input type="checkbox" name={name} defaultChecked={defaultChecked} className="mt-0.5 h-5 w-5 accent-maroon" />
      <span>
        <span className="font-semibold text-charcoal">{label}</span>
        {hint && <span className="block text-xs text-ink-soft">{hint}</span>}
      </span>
    </label>
  );
}

export function PageTitle({ title, description, children }: { title: string; description?: string; children?: ReactNode }) {

import { cn } from "@/lib/utils";

/**
 * Original geometric mandala (pure SVG, no external assets).
 * Decorative only â†’ aria-hidden. Uses currentColor so callers set the tone.
 */
export function Mandala({
  className,
  rings = 3,
  spin = false,
}: {
  className?: string;
  rings?: 1 | 2 | 3;
  spin?: boolean;
}) {
  const petals = Array.from({ length: 16 }, (_, i) => i * (360 / 16));
  const inner = Array.from({ length: 8 }, (_, i) => i * 45 + 22.5);
  const ticks = Array.from({ length: 72 }, (_, i) => i * 5);

  return (
    <svg
      viewBox="-100 -100 200 200"
      className={cn("text-gold", spin && "spin-slow", className)}
      fill="none"
      stroke="currentColor"
      strokeWidth="0.6"
      aria-hidden="true"
      focusable="false"
    >
      <circle r="97" strokeWidth="0.8" />
      <circle r="94" strokeOpacity="0.5" />
      {ticks.map((a) => (
        <line key={a} x1="0" y1="-94" x2="0" y2={a % 15 === 0 ? -89 : -91.5} transform={`rotate(${a})`} strokeOpacity="0.7" />
      ))}

      {rings >= 3 &&
        petals.map((a) => (
          <path key={`o${a}`} d="M0 -86 C 7 -76, 7 -66, 0 -58 C -7 -66, -7 -76, 0 -86 Z" transform={`rotate(${a})`} />
        ))}
      <circle r="56" strokeOpacity="0.6" />

      {rings >= 2 &&
        petals.map((a) => (
          <path key={`m${a}`} d="M0 -54 C 9 -46, 9 -36, 0 -28 C -9 -36, -9 -46, 0 -54 Z" transform={`rotate(${a + 11.25})`} strokeOpacity="0.9" />
        ))}
      <circle r="27" />

      {inner.map((a) => (
        <path key={`i${a}`} d="M0 -26 C 5 -20, 5 -14, 0 -9 C -5 -14, -5 -20, 0 -26 Z" transform={`rotate(${a})`} />
      ))}
      {/* eight-point star */}
      <path d="M0 -17 L5 -5 L17 0 L5 5 L0 17 L-5 5 L-17 0 L-5 -5 Z" strokeWidth="0.8" />
      <circle r="4.5" strokeWidth="0.9" />
      <circle r="1.6" fill="currentColor" stroke="none" />
    </svg>
  );
}

  return (
    <div className={cn("prose-ka text-ink-soft", className)}>
      {blocks.map((b, i) => {

          const [head, ...rest] = b.split("\n");

              <h2>{head.slice(3)}</h2>
              {rest.length > 0 && <p className="whitespace-pre-line">{rest.join("\n")}</p>}
            </div>
          );
        }
        return (
          <p key={i} className="whitespace-pre-line">
            {b}
          </p>
        );
      })}
    </div>
  );
}

// Pure slot logic (no framework imports) so it is easy to unit-test.
// Business time zone: Asia/Kolkata = UTC+05:30, no daylight saving.

export const TZ = "Asia/Kolkata";
const IST_OFFSET = "+05:30";

export interface SlotRule {
  weekday: number; // 0 = Sunday
  start_time: string; // "HH:MM" or "HH:MM:SS"
  end_time: string;
  is_active: boolean;
}

export interface SlotUnavailable {
  starts_at: string;
  ends_at: string;
}

export interface SlotSettings {
  slot_minutes: number;
  buffer_minutes: number;
  min_notice_hours: number;
  max_advance_days: number;
}

export interface Slot {
  startsAt: string; // ISO (UTC)
  label: string; // "7:00 PM"
}

export const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;

/** IST wall-clock date + "HH:MM" â†’ absolute Date. */
export function istToDate(date: string, hhmm: string): Date {
  return new Date(`${date}T${hhmm.slice(0, 5)}:00${IST_OFFSET}`);
}

/** Today's (or any instant's) calendar date in IST as YYYY-MM-DD. */
export function istDateString(d: Date): string {
  const parts = new Intl.DateTimeFormat("en-CA", {
    timeZone: TZ,
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
  }).format(d);
  return parts; // en-CA â†’ YYYY-MM-DD
}

export function addDays(date: string, n: number): string {
  const d = new Date(`${date}T00:00:00Z`);
  d.setUTCDate(d.getUTCDate() + n);
  return d.toISOString().slice(0, 10);
}

/** Weekday (0=Sun) of a calendar date; independent of time zone. */
export function weekdayOf(date: string): number {
  return new Date(`${date}T00:00:00Z`).getUTCDay();
}

const toMinutes = (t: string): number => {
  const [h, m] = t.split(":");
  return Number(h) * 60 + Number(m);
};

const pad = (n: number) => String(n).padStart(2, "0");

export function formatTimeIst(d: Date): string {
  return new Intl.DateTimeFormat("en-IN", {
    timeZone: TZ,
    hour: "numeric",
    minute: "2-digit",
    hour12: true,
  })
    .format(d)
    .toUpperCase();
}

export function formatDateTimeIst(iso: string): string {
  return new Intl.DateTimeFormat("en-IN", {
    timeZone: TZ,
    weekday: "short",
    day: "numeric",
    month: "short",
    year: "numeric",
    hour: "numeric",
    minute: "2-digit",
    hour12: true,
  }).format(new Date(iso));
}

export function formatDateIst(date: string): string {
  return new Intl.DateTimeFormat("en-IN", {
    timeZone: "UTC",
    weekday: "long",
    day: "numeric",
    month: "long",
    year: "numeric",
  }).format(new Date(`${date}T00:00:00Z`));
}

export function generateSlots(args: {
  date: string; // YYYY-MM-DD (IST)
  durationMinutes: number;
  rules: SlotRule[];
  settings: SlotSettings;
  unavailable: SlotUnavailable[];
  now?: Date;
}): Slot[] {
  const { date, durationMinutes, rules, settings, unavailable } = args;
  const now = args.now ?? new Date();
  if (!DATE_RE.test(date)) return [];

  const rule = rules.find((r) => r.weekday === weekdayOf(date) && r.is_active);
  if (!rule) return [];

  const earliest = now.getTime() + settings.min_notice_hours * 3_600_000;
  const latest = now.getTime() + (settings.max_advance_days + 1) * 86_400_000;
  const step = Math.max(5, settings.slot_minutes);
  const startMin = toMinutes(rule.start_time);
  const endMin = toMinutes(rule.end_time);
  const busy = unavailable.map((u) => ({
    s: new Date(u.starts_at).getTime(),
    e: new Date(u.ends_at).getTime(),
  }));

  const slots: Slot[] = [];
  for (let m = startMin; m + durationMinutes <= endMin; m += step) {
    const start = istToDate(date, `${pad(Math.floor(m / 60))}:${pad(m % 60)}`);
    const t = start.getTime();
    if (t < earliest || t > latest) continue;
    const held = t + (durationMinutes + settings.buffer_minutes) * 60_000;
    if (busy.some((b) => t < b.e && held > b.s)) continue;
    slots.push({ startsAt: start.toISOString(), label: formatTimeIst(start) });
  }
  return slots;
}

/** Candidate dates (IST) in the booking window whose weekday has an active rule. */
export function candidateDates(rules: SlotRule[], settings: SlotSettings, now = new Date()): string[] {
  const active = new Set(rules.filter((r) => r.is_active).map((r) => r.weekday));
  const today = istDateString(now);
  const out: string[] = [];
  for (let i = 0; i <= settings.max_advance_days; i++) {
    const d = addDays(today, i);
    if (active.has(weekdayOf(d))) out.push(d);
  }
  return out;
}

/**
 * Session-aware client (reads the signed-in user's cookies). Uses the public
 * key, so every query is subject to Row Level Security as that user.
 * Admin CRUD uses this client â€” RLS (is_admin()) is the real enforcement.
 */
export async function createClient() {
  const cookieStore = await cookies();
  return createServerClient(supabaseUrl!, supabaseKey!, {
    cookies: {
      getAll() {
        return cookieStore.getAll();
      },
      setAll(list) {
        try {
          list.forEach(({ name, value, options }) => cookieStore.set(name, value, options));
        } catch {
          // Called from a Server Component: cookies are refreshed in proxy.ts instead.
        }
      },
    },
  });
}

 * cookies(), pages that use it can be statically generated / revalidated.
 */
export function createPublicClient() {
  return createSupabaseClient(supabaseUrl!, supabaseKey!, {
    auth: { persistSession: false, autoRefreshToken: false },
  });

  if (row) {

    const m = media as { storage_path?: string; alt_text?: string } | null | undefined;
    return {
      key: String(row.section_key),
      eyebrow: (row.eyebrow as string | null) ?? null,
      heading: (row.heading as string | null) ?? null,
      body: (row.body as string | null) ?? null,
      items: Array.isArray(row.items) ? (row.items as SectionItem[]) : [],
      ctaLabel: (row.cta_label as string | null) ?? null,
      ctaHref: (row.cta_href as string | null) ?? null,
      imageUrl: mediaPublicUrl(m?.storage_path),
      imageAlt: m?.alt_text ?? "",
      imageAssetId: (row.image_asset_id as string | null) ?? null,
      sortOrder: Number(row.sort_order ?? 0),
      isVisible: Boolean(row.is_visible),
      isDraft: Boolean(row.is_draft),
      isDefault: false,
    };
  }
  const d = def!;
  return {
    key: d.key,
    eyebrow: d.eyebrow ?? null,
    heading: d.heading ?? null,
    body: d.body ?? null,
    items: d.items ?? [],
    ctaLabel: d.ctaLabel ?? null,
    ctaHref: d.ctaHref ?? null,
    imageUrl: null,
    imageAlt: "",
    imageAssetId: null,
    sortOrder: d.sortOrder,
    isVisible: d.isVisible ?? true,
    isDraft: d.draft ?? false,
    isDefault: true,
  };
}

function fromDefaults(def: DefaultPage | undefined, slug: string) {
  return {
    notFound: false as const,
    meta: {
      slug,
      title: def?.title ?? slug,
      metaTitle: def?.metaTitle ?? null,
      metaDescription: def?.metaDescription ?? null,
      isPublished: true,
      isDraft: def?.draft ?? false,
    } satisfies PageMeta,
    sections: (def?.sections ?? []).map((s) => toSection(s)).sort((a, b) => a.sortOrder - b.sortOrder),
  };
}

export type PageContent =
  | { notFound: true }
  | { notFound: false; meta: PageMeta; sections: Section[] };

export const getPageContent = cache(async (slug: string): Promise<PageContent> => {
  const def = getDefaultPage(slug);
  if (!isSupabaseConfigured) return fromDefaults(def, slug);

  try {
    const sb = createPublicClient();
    const { data: page, error } = await sb
      .from("pages")
      .select("id, slug, title, meta_title, meta_description, is_published, is_draft")
      .eq("slug", slug)
      .maybeSingle();
    if (error) throw error;

    if (!page) {
      // RLS hides unpublished pages: distinguish "unpublished" from "never created".
      const { data: exists } = await sb.rpc("page_exists", { p_slug: slug });
      if (exists) return { notFound: true };
      return fromDefaults(def, slug);
    }

    const { data: rows, error: secErr } = await sb
      .from("page_sections")
      .select("*, media:media_assets(storage_path, alt_text)")
      .eq("page_id", page.id);
    if (secErr) throw secErr;

    const byKey = new Map<string, Section>();
    for (const d of def?.sections ?? []) byKey.set(d.key, toSection(d));
    for (const r of rows ?? []) byKey.set(String(r.section_key), toSection(undefined, r));

    // RLS hides hidden sections from anon, so a built-in default for the same key
    // would reappear. Ask the database which keys the owner has hidden and drop them.
    const { data: hidden } = await sb.rpc("hidden_section_keys", { p_slug: slug });
    for (const key of (hidden as string[] | null) ?? []) byKey.delete(key);
    const sections = [...byKey.values()].sort((a, b) => a.sortOrder - b.sortOrder);

    return {
      notFound: false,
      meta: {
        slug: page.slug,
        title: page.title,
        metaTitle: page.meta_title,
        metaDescription: page.meta_description,
        isPublished: page.is_published,
        isDraft: page.is_draft,
      },
      sections,
    };
  } catch (e) {
    logError(`page:${slug}`, e);
    return fromDefaults(def, slug);
  }
});

export function sectionByKey(sections: Section[], key: string): Section | undefined {
  return sections.find((s) => s.key === key);
}

});

export function cn(...parts: Array<string | false | null | undefined>): string {
  return parts.filter(Boolean).join(" ");
}

export function slugify(input: string): string {
  return input
    .toLowerCase()
    .normalize("NFKD")
    .replace(/[Ì€-Í¯]/g, "")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "")
    .slice(0, 80);
}

/** 150000 paise â†’ "â‚¹1,500". null â†’ null (caller decides how to word "price on request"). */
export function formatPrice(paise: number | null | undefined, currency = "INR"): string | null {
  if (paise === null || paise === undefined) return null;
  return new Intl.NumberFormat("en-IN", {
    style: "currency",
    currency,
    maximumFractionDigits: paise % 100 === 0 ? 0 : 2,
  }).format(paise / 100);
}

/** "1500" or "1,500.50" â†’ paise (integer). Returns null for blank, NaN for invalid. */
export function rupeesToPaise(input: string | null | undefined): number | null {
  const v = (input ?? "").replace(/[,\sâ‚¹]/g, "").trim();
  if (v === "") return null;
  if (!/^\d+(\.\d{1,2})?$/.test(v)) return Number.NaN;
  return Math.round(parseFloat(v) * 100);
}

export function paiseToRupeesInput(paise: number | null | undefined): string {
  if (paise === null || paise === undefined) return "";
  return paise % 100 === 0 ? String(paise / 100) : (paise / 100).toFixed(2);
}

export function formatDuration(minutes: number): string {
  if (minutes < 60) return `${minutes} min`;
  const h = Math.floor(minutes / 60);
  const m = minutes % 60;
  return m === 0 ? `${h} hr` : `${h} hr ${m} min`;
}

/** Only allow same-site paths or http(s)/mailto/tel URLs from editable content. */
export function safeHref(href: string | null | undefined, fallback = "#"): string {
  if (!href) return fallback;
  const v = href.trim();
  if (v.startsWith("/") && !v.startsWith("//")) return v;
  if (/^(https?:\/\/|mailto:|tel:)/i.test(v)) return v;
  return fallback;
}

export function whatsappLink(number: string | null | undefined, text?: string): string | null {
  const digits = (number ?? "").replace(/\D/g, "");
  if (!digits) return null;
  return `https://wa.me/${digits}${text ? `?text=${encodeURIComponent(text)}` : ""}`;
}

/** Lines "Title | Text" â†’ items. Lines without a pipe become title-only. */
export function parseItems(raw: string): Array<{ title: string; text: string }> {
  return raw
    .split("\n")
    .map((l) => l.trim())
    .filter(Boolean)
    .map((line) => {
      const i = line.indexOf("|");
      return i === -1
        ? { title: line, text: "" }
        : { title: line.slice(0, i).trim(), text: line.slice(i + 1).trim() };
    });
}

export function itemsToText(items: Array<{ title: string; text: string }>): string {
  return items.map((i) => (i.text ? `${i.title} | ${i.text}` : i.title)).join("\n");
}

export function parseLines(raw: string): string[] {
  return raw
    .split("\n")
    .map((l) => l.trim())
    .filter(Boolean);
}
